
pipeline {
    agent any

    environment {
        EC2_PUBLIC_IP = '16.170.236.50'
    }

    stages {

        stage('Connect to AWS EC2') {
            steps {
                withCredentials([
                    sshUserPrivateKey(
                        credentialsId: 'aws-cred',
                        keyFileVariable: 'SSH_KEY',
                        usernameVariable: 'SSH_USER'
                    )
                ]) {
                    bat '''
                        icacls "%SSH_KEY%" /inheritance:r
                        icacls "%SSH_KEY%" /grant:r "*S-1-5-18:R"
                        icacls "%SSH_KEY%" /remove:g "*S-1-5-32-545"

                        if errorlevel 1 exit /b 1

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "echo Connected to EC2 && whoami && hostname"

                        if errorlevel 1 exit /b 1
                    '''
                }
            }
        }

        stage('Install Docker on AWS EC2 Instance') {
            steps {
                withCredentials([
                    sshUserPrivateKey(
                        credentialsId: 'aws-cred',
                        keyFileVariable: 'SSH_KEY',
                        usernameVariable: 'SSH_USER'
                    )
                ]) {
                    bat '''
                        icacls "%SSH_KEY%" /inheritance:r
                        icacls "%SSH_KEY%" /grant:r "*S-1-5-18:R"
                        icacls "%SSH_KEY%" /remove:g "*S-1-5-32-545"

                        if errorlevel 1 exit /b 1

                        echo Updating Ubuntu...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo apt update"
                        if errorlevel 1 exit /b 1

                        echo Installing prerequisites...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo apt install -y ca-certificates curl"
                        if errorlevel 1 exit /b 1

                        echo Creating Docker keyring directory...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo install -m 0755 -d /etc/apt/keyrings"
                        if errorlevel 1 exit /b 1

                        echo Downloading Docker GPG key...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc"
                        if errorlevel 1 exit /b 1

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo chmod a+r /etc/apt/keyrings/docker.asc"
                        if errorlevel 1 exit /b 1

                        echo Creating Docker repository file...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "echo 'Types: deb' | sudo tee /etc/apt/sources.list.d/docker.sources"
                        if errorlevel 1 exit /b 1

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "echo 'URIs: https://download.docker.com/linux/ubuntu' | sudo tee -a /etc/apt/sources.list.d/docker.sources"
                        if errorlevel 1 exit /b 1

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "echo 'Suites: noble' | sudo tee -a /etc/apt/sources.list.d/docker.sources"
                        if errorlevel 1 exit /b 1

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "echo 'Components: stable' | sudo tee -a /etc/apt/sources.list.d/docker.sources"
                        if errorlevel 1 exit /b 1

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "echo 'Architectures: amd64' | sudo tee -a /etc/apt/sources.list.d/docker.sources"
                        if errorlevel 1 exit /b 1

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "echo 'Signed-By: /etc/apt/keyrings/docker.asc' | sudo tee -a /etc/apt/sources.list.d/docker.sources"
                        if errorlevel 1 exit /b 1

                        echo Updating package index...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo apt update"
                        if errorlevel 1 exit /b 1

                        echo Installing Docker packages...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin"
                        if errorlevel 1 exit /b 1

                        echo Starting Docker service...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo systemctl enable --now docker"
                        if errorlevel 1 exit /b 1

                        echo Granting Docker group permission...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo usermod -aG docker %SSH_USER%"
                        if errorlevel 1 exit /b 1

                        echo Docker installation completed.
                    '''
                }
            }
        }

        stage('Verify Docker Installation') {
            steps {
                withCredentials([
                    sshUserPrivateKey(
                        credentialsId: 'aws-cred',
                        keyFileVariable: 'SSH_KEY',
                        usernameVariable: 'SSH_USER'
                    )
                ]) {
                    bat '''
                        icacls "%SSH_KEY%" /inheritance:r
                        icacls "%SSH_KEY%" /grant:r "*S-1-5-18:R"
                        icacls "%SSH_KEY%" /remove:g "*S-1-5-32-545"

                        if errorlevel 1 exit /b 1

                        echo Checking Docker version...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo docker --version"
                        if errorlevel 1 exit /b 1

                        echo Checking Docker service...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo systemctl is-active docker"
                        if errorlevel 1 exit /b 1

                        echo Checking Docker Compose...

                        ssh -i "%SSH_KEY%" -o StrictHostKeyChecking=accept-new %SSH_USER%@%EC2_PUBLIC_IP% "sudo docker compose version"
                        if errorlevel 1 exit /b 1

                        echo All Docker verification checks completed successfully.
                    '''
                }
            }
        }
    }
}
