pipeline {
    agent any

    environment {
        JAVA_HOME = 'C:\\Program Files\\Java\\jdk-21.0.12.1'
        MAVEN = 'C:\\ProgramData\\Jenkins\\.jenkins\\tools\\hudson.tasks.Maven_MavenInstallation\\Maven\\bin\\mvn.cmd'
        PATH = "${JAVA_HOME}\\bin;${env.PATH}"
    }

    stages {
        stage('Clone Repository') {
            steps {
                git url: 'https://github.com/SawanPanwar/ORSProject04.git',
                    branch: 'main'
            }
        }

        stage('Build New WAR') {
            steps {
                bat 'call "%MAVEN%" clean package'
            }
        }

        stage('Build Web and Pull MySQL') {
            steps {
                bat 'docker-compose build web'
                bat 'docker-compose pull db'
            }
        }

        stage('Docker Tagging') {
            steps {
                bat 'docker tag web:latest prernathakur/web:latest'
                bat 'docker tag mysql:8.0 prernathakur/mysql:8.0'
            }
        }

        stage('Docker Login') {
            steps {
                withCredentials([
                    usernamePassword(
                        credentialsId: 'dockerhub-creds',
                        usernameVariable: 'DH_USER',
                        passwordVariable: 'DH_PASS'
                    )
                ]) {
                    bat 'docker login -u %DH_USER% -p %DH_PASS%'
                }
            }
        }

        stage('Docker Push') {
            steps {
                bat 'docker push prernathakur/web:latest'
                bat 'docker push prernathakur/mysql:8.0'
            }
        }
    }
}
